// Balloons const emojis = ['🎈','🎈','🎉','🎊','💛','🌟','🎈']; const balloonsEl = document.getElementById('balloons'); for (let i = 0; i < 10; i++) { const b = document.createElement('div'); b.className = 'balloon'; b.textContent = emojis[Math.floor(Math.random()*emojis.length)]; b.style.cssText = left:${Math.random()*95}%;--dur:${(Math.random()*5+6).toFixed(1)}s;--delay:${(Math.random()*8).toFixed(1)}s;font-size:${(Math.random()*1.5+2).toFixed(1)}rem; balloonsEl.appendChild(b); } // Confetti const canvas = document.getElementById('confettiCanvas'); const ctx = canvas.getContext('2d'); let W, H, particles = []; function resize() { W = canvas.width = window.innerWidth; H = canvas.height = window.innerHeight; } window.addEventListener('resize', resize); resize(); const COLORS = ['#FFD700','#FF6B35','#FF3CAC','#2BD2FF','#fff','#a8ff78','#ff8c00']; function createParticle(x, y) { return { x, y, vx: (Math.random() - 0.5) * 10, vy: (Math.random() * -12) - 3, size: Math.random() * 8 + 4, color: COLORS[Math.floor(Math.random() * COLORS.length)], rotation: Math.random() * 360, spin: (Math.random() - 0.5) * 8, gravity: 0.35, alpha: 1, shape: Math.random() > 0.5 ? 'rect' : 'circle' }; } function burst() { const cx = W / 2, cy = H / 2; for (let i = 0; i < 140; i++) particles.push(createParticle(cx + (Math.random()-0.5)*60, cy + (Math.random()-0.5)*60)); if (!animating) loop(); } let animating = false; function loop() { animating = true; ctx.clearRect(0, 0, W, H); particles.forEach(p => { p.x += p.vx; p.y += p.vy; p.vy += p.gravity; p.rotation += p.spin; p.alpha -= 0.012; ctx.save(); ctx.globalAlpha = Math.max(0, p.alpha); ctx.translate(p.x, p.y); ctx.rotate(p.rotation * Math.PI / 180); ctx.fillStyle = p.color; if (p.shape === 'circle') { ctx.beginPath(); ctx.arc(0, 0, p.size/2, 0, Math.PI*2); ctx.fill(); } else { ctx.fillRect(-p.size/2, -p.size/4, p.size, p.size/2); } ctx.restore(); }); particles = particles.filter(p => p.alpha > 0 && p.y < H + 20); if (particles.length) requestAnimationFrame(loop); else animating = false; } setTimeout(() => burst(), 600); setTimeout(() => burst(), 1400);
